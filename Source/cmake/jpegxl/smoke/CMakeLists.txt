cmake_minimum_required(VERSION 3.25)
project(vove_jpegxl_smoke LANGUAGES CXX)
find_package(vove-jpegxl-decode CONFIG REQUIRED)
add_executable(vove-jpegxl-smoke main.cpp)
add_executable(vove-jpegxl-baseline baseline.cpp)
target_link_libraries(vove-jpegxl-smoke PRIVATE vove-jpegxl-decode::jxl_dec)
foreach(target vove-jpegxl-smoke vove-jpegxl-baseline)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
  target_link_options(${target} PRIVATE -static -Wl,--gc-sections -Wl,--strip-all)
endforeach()
